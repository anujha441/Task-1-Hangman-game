import random

words = ["apple", "tiger", "phone", "water", "chair"]
word = random.choice(words)

guessed = ""
chances = 6

while chances > 0:
    display = ""

    for letter in word:
        if letter in guessed:
            display += letter
        else:
            display += "_"

    print("Word:", display)

    if display == word:
        print("You Win!")
        break

    guess = input("a: ")
    guessed += guess

    if guess not in word:
        chances -= 1
        print("Wrong! Chances left:", chances)

if chances == 0:
    print("Game Over!")
    print("Word was:", word)
